104void UStartWidget::UpdatePlayerList(const TArray<FString>& playerNames)
105{
106 PRINTLOG(TEXT("UStartUI::UpdatePlayerList called with %d player(s)."), playerNames.Num());
107 if (!playerList) return;
108
109 playerList->ClearChildren(); // Clear existing entries
110 playerList->SetVisibility(ESlateVisibility::Visible);

124 // 내 PlayerIndex 구하기 - 분할 화면/멀티플레이 모두 지원되도록 Owning Player 기준으로 조회한다.
125 int32 LocalPlayerIndex = INDEX_NONE;
126 const AYiSanPlayerState* LocalPlayerState = GetOwningPlayerState<AYiSanPlayerState>();
127
128 if (!LocalPlayerState)
129 {
130 if (const APlayerController* PlayerController = GetOwningPlayer())
131 {
132 LocalPlayerState = PlayerController->GetPlayerState<AYiSanPlayerState>();
133 }
134 }
135
136 if (LocalPlayerState)
137 {
138 LocalPlayerIndex = LocalPlayerState->PlayerIndex;
139 PRINTLOG(TEXT("LocalPlayerIndex = %d"), LocalPlayerIndex);
140 }
141 else
142 {
143 PRINTLOG(TEXT("Failed to resolve LocalPlayerIndex. Using INDEX_NONE."));
144 }
145
146 for (const FString& PlayerInfoString : playerNames)
147 {
148 TArray<FString> PlayerInfo;
149 PlayerInfoString.ParseIntoArray(PlayerInfo, TEXT(":"), true);
150
151 if (PlayerInfo.Num() == 2)
152 {
153 int32 PlayerIndex = FCString::Atoi(*PlayerInfo[0]);
154 FString PlayerName = PlayerInfo[1];
155
156 UPlayerListItem* PlayerListItem = CreatePlayerListItem(PlayerIndex, LocalPlayerIndex, PlayerName);
157 if (PlayerListItem)
158 {
159 playerList->AddChild(PlayerListItem);
160 }
161 }
162 }
