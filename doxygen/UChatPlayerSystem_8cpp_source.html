78void UChatPlayerSystem::MulticastRPC_AddChatMessage_Implementation(const FChatMessage& ChatMessage)
79{
80 // 현재 클라이언트의 로컬 플레이어 컨트롤러를 가져옴 (멀티플레이 환경에서도 각 클라에 1개만 존재)
81 // 없으면 UI 접근 불가 → 조기 종료
82 auto* PC = GetWorld()->GetFirstPlayerController();
83 if (!PC)
84 return;
85
86 // 로컬 컨트롤러가 조종 중인 Pawn을 APlayerActor 타입으로 변환
87 // 변환 실패 시 종료 (월드 내 Pawn이 없거나 아직 스폰 전일 수 있음)
88 auto* Player = Cast<APlayerActor>(PC->GetPawn());
89 if (!Player)
90 return;
91
92 // 플레이어가 보유한 채팅 UI 위젯 참조
93 // UI 위젯이 생성되지 않았으면 종료
94 auto* ChatBox = Player->ChatBoxWidget.Get();
95 if (!ChatBox)
96 return;
97
98 // 로컬 클라이언트의 채팅창에 서버에서 전달받은 메시지를 출력
99 ChatBox->AddChatMessage(ChatMessage);
100}

103void UChatPlayerSystem::AnnouncePlayerJoin()
104{
105 if (!IsValid(Owner) || !IsValid(ChatBoxWidget))
106 return;
107
108 const FString Message = FString::Printf(TEXT("%s 전하 어가, 행궁에 이르셨습니다."), *Owner->GetPlayerDisplayName());
109 const FChatMessage ChatMessage(EChatMessageType::System, -1, GameString::System, Message);
110 ServerRPC_SendChatMessage(ChatMessage);
111
112 if (auto GS = GetWorld()->GetGameState<AYisanGameState>())
113 {
114 GS->ServerRPC_PlaySound(EGameSoundType::Enter_Game);
115 }
116
117 // 플레이어 입장 메시지 표시 시점에 로딩 화면 숨김 (게스트 join 시점 처리)
118 PRINTLOG(TEXT("[LOADING_FLOW] AnnouncePlayerJoin - Player entered game, hiding loading screen"));
119 if (Owner)
120 {
121 if (auto PC = Cast<APlayerControl>(Owner->GetController()))
122 {
123 if (PC->IsLocalController())
124 {
125 if (auto TM = ULoadingTransitionManager::Get(PC))
126 {
127 TM->HideLoadingScreen();
128 }
129 }
130 }
131 }
