91void UStateWidget::UpdateSpectrumVisual(float DeltaTime)
92{
93 if ( !SpectrumProgressBar->IsVisible())
94 return;
95
96 const float Delta = FMath::Max(DeltaTime, 0.0f);
97 const float TargetValue = SpectrumDisplayValue;
98 const float CurrentValue = SpectrumProgressBar->GetPercent();
99 const float InterpSpeed = TargetValue > CurrentValue ? SpectrumRiseSpeed : SpectrumDecaySpeed;
100 const float NewPercent = FMath::FInterpConstantTo(CurrentValue, TargetValue, Delta, InterpSpeed);
101 SpectrumProgressBar->SetPercent(FMath::Clamp(NewPercent, 0.0f, 1.0f));
102}

128void UStateWidget::OnUpdateQuest(EBuildingType InBuildingType)
129{
130 if (!QuestTargetText || !QuestTargetImage || !QuestArrowImage)
131 return;
132
133 if (InBuildingType != EBuildingType::None)
134 LastQuestTarget = InBuildingType;
135
136 const bool bDisplayLastByeolju = (InBuildingType == EBuildingType::None && LastQuestTarget == EBuildingType::Byeolju);
137 const EBuildingType DisplayType = bDisplayLastByeolju ? LastQuestTarget : InBuildingType;
138
139 QuestDisplayType = DisplayType;
140
141 if (DisplayType == EBuildingType::None)
142 {
143 QuestTargetText->SetText(FText::GetEmpty());
144 QuestTargetText->SetVisibility(ESlateVisibility::Hidden);
145 QuestTargetImage->SetVisibility(ESlateVisibility::Hidden);
146 QuestArrowImage->SetVisibility(ESlateVisibility::Hidden);
147 return;
148 }
149
150 auto GameDataManager = UGameDataManager::Get(GetWorld());
151 auto BuildingName = GameDataManager->GetBuildingDataName(DisplayType);
152 QuestTargetText->SetText(FText::FromString(BuildingName));
153 QuestTargetText->SetVisibility(ESlateVisibility::Visible);
154 QuestArrowImage->SetVisibility(ESlateVisibility::Visible);
155 QuestArrowImage->SetVisibility(bDisplayLastByeolju ? ESlateVisibility::Hidden : ESlateVisibility::Visible);
156
157 FBuildingAssetData AssetData;
158 if ( GameDataManager->GetBuildingAssetData(DisplayType, AssetData) )
159 {
160 UBuildingDetailData* DetailAsset = AssetData.BuildingDetailDataAsset.LoadSynchronous();
161 if ( DetailAsset )
162 {
163 TSoftObjectPtr<UTexture2D> LoadedTexture;
164 if ( DetailAsset->LoadThumbnailImage(LoadedTexture) && LoadedTexture.Get() )
165 {
166 QuestTargetImage->SetBrushFromTexture(LoadedTexture.Get());
167 QuestTargetImage->SetVisibility(ESlateVisibility::Visible);
168 }
169 }
170 }
171}

173void UStateWidget::UpdateQuestArrow()
174{
175 // 퀘스트가 없으면 화살표 숨김
176 if (QuestDisplayType == EBuildingType::None || !QuestArrowImage)
177 return;
178
179 // 플레이어 컨트롤러 가져오기
180 APlayerController* PC = UGameplayStatics::GetPlayerController(GetWorld(), 0);
181 if (!PC)
182 return;
183
184 APawn* PlayerPawn = PC->GetPawn();
185 if (!PlayerPawn)
186 return;
187
188 // 목표 건물 찾기
189 ABuilding* TargetBuilding = nullptr;
190 TArray<AActor*> FoundBuildings;
191 UGameplayStatics::GetAllActorsOfClass(GetWorld(), ABuilding::StaticClass(), FoundBuildings);
192
193 for (AActor* Actor : FoundBuildings)
194 {
195 ABuilding* Building = Cast<ABuilding>(Actor);
196 if (Building && Building->BuildingType == QuestDisplayType)
197 {
198 TargetBuilding = Building;
199 break;
200 }
201 }
202
203 if (!TargetBuilding)
204 return;
205
206 // 플레이어 위치와 목표 건물 위치
207 FVector PlayerLocation = PlayerPawn->GetActorLocation();
208 FVector TargetLocation = TargetBuilding->GetActorLocation();
209
210 // 플레이어에서 목표로의 방향 벡터 (2D 평면)
211 FVector DirectionToTarget = TargetLocation - PlayerLocation;
212 DirectionToTarget.Z = 0; // 수평 방향만 고려
213 DirectionToTarget.Normalize();
214
215 // 플레이어의 정면 방향 (카메라 기준)
216 FVector CameraForward = PC->GetControlRotation().Vector();
217 CameraForward.Z = 0;
218 CameraForward.Normalize();
219
220 // 두 벡터 사이의 각도 계산 (Degree)
221 float DotProduct = FVector::DotProduct(CameraForward, DirectionToTarget);
222 float CrossProduct = FVector::CrossProduct(CameraForward, DirectionToTarget).Z;
223
224 // 각도 계산 (-180 ~ 180)
225 float AngleDegrees = FMath::RadiansToDegrees(FMath::Acos(FMath::Clamp(DotProduct, -1.0f, 1.0f)));
226
227 // CrossProduct의 부호로 방향 결정
228 if (CrossProduct < 0)
229 AngleDegrees = -AngleDegrees;
230
231 // 화살표 회전 (화살표 머리가 12시 방향이므로 그대로 적용)
232 QuestArrowImage->SetRenderTransformAngle(AngleDegrees);
233}
