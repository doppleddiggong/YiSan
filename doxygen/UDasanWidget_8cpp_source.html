34void UDasanWidget::UpdateDasanState(EDasanState MainState, ETourState TourState, EAnswerState AnswerState,
35 const FString& BuildingName, const FString& PlayerName, int32 CurrentPlayers, int32 MaxPlayers)
36{
37 if (!Text_Msg)
38 {
39 PRINTLOG(TEXT("[DasanWidget] UpdateDasanState - Text_Msg is nullptr!"));
40 return;
41 }
42
43 FString StateMessage = GetStateMessage(MainState, TourState, AnswerState, BuildingName, PlayerName, CurrentPlayers, MaxPlayers);
44 FString ResourcePath = GetResourcePath(MainState, TourState, AnswerState);
45
46 PRINTLOG(TEXT("[DasanWidget] UpdateDasanState - MainState: %s, TourState: %s, AnswerState: %s, Message: %s"),
47 *ENUM_TO_NAME(EDasanState, MainState),
48 *ENUM_TO_NAME(ETourState, TourState),
49 *ENUM_TO_NAME(EAnswerState, AnswerState),
50 *StateMessage);
51
52 Text_Msg->SetText(FText::FromString(StateMessage));
53
54 // 텍스처 로드 (올바른 경로 형식 사용)
55 if (UTexture2D* Texture = LoadObject<UTexture2D>(nullptr, *ResourcePath))
56 {
57 Image_Icon->SetBrushFromTexture(Texture);
58 }
59 else
60 {
61 PRINTLOG(TEXT("[DasanWidget] 텍스처 로드 실패: %s"), *ResourcePath);
62 }
63}

65void UDasanWidget::ShowExplainDialog(const FString& ExplainText)
66{
67 if (!Text_Explain || !Border_Explain)
68 {
69 PRINTLOG(TEXT("[DasanWidget] ShowExplainDialog - Text_Explain or Border_Explain is nullptr!"));
70 return;
71 }
72
73 PRINTLOG(TEXT("[DasanWidget] ShowExplainDialog - Showing: %s"), *ExplainText);
74
75 Text_Explain->SetText(FText::FromString(ExplainText));
76 Border_Explain->SetVisibility(ESlateVisibility::Visible);
77
78 // 기존 타이머 클리어
79 if (GetWorld() && GetWorld()->GetTimerManager().IsTimerActive(ExplainDialogTimerHandle))
80 GetWorld()->GetTimerManager().ClearTimer(ExplainDialogTimerHandle);
81
82 // 3초 후 자동 숨김
83 if (GetWorld())
84 {
85 GetWorld()->GetTimerManager().SetTimer(
86 ExplainDialogTimerHandle,
87 this,
88 &UDasanWidget::HideExplainDialog,
89 3.0f,
90 false
91 );
92 }
93}

115FString UDasanWidget::GetStateMessage(EDasanState MainState, ETourState TourState, EAnswerState AnswerState,
116 const FString& BuildingName, const FString& PlayerName, int32 CurrentPlayers, int32 MaxPlayers) const
117{
118 switch (MainState)
119 {
120 case EDasanState::Tour:
121 {
122 switch (TourState)
123 {
124 case ETourState::TourMove:
125 // "%s 를 향해 이동 중입니다"
126 if (!BuildingName.IsEmpty())
127 return FString::Printf(TEXT("%s 으로 이동하시죠"), *BuildingName);
128 return TEXT("이동중");
129
130 case ETourState::TourWait:
131 return TEXT("집결대기중");
132
133 case ETourState::TourExplain:
134 // "%s 에 대한 설명을 진행합니다"
135 if (!BuildingName.IsEmpty())
136 return FString::Printf(TEXT("%s"), *BuildingName);
137 return TEXT("설명중");
138
139 case ETourState::TourEnd:
140 return TEXT(""); // 숨김
141 default:
142 return TEXT("");
143 }
144 }
145
146 case EDasanState::Answer:
147 {
148 switch (AnswerState)
149 {
150 case EAnswerState::AnswerListen:
151 // "%s 군주님의 질문을 듣고 있습니다"
152 if (!PlayerName.IsEmpty())
153 return FString::Printf(TEXT("%s 군주님의 질문을 듣고 있습니다"), *PlayerName);
154 return TEXT("경청중");
155
156 case EAnswerState::AnswerReply:
157 return TEXT("답변중");
158 case EAnswerState::AnswerEnd:
159 return TEXT(""); // 숨김
160 default:
161 return TEXT("");
162 }
163 }
164
165 default:
166 return TEXT("");
167 }
168}
