99bool UAnswerStateSystem::CanStartAnswer(const FString& PlayerName, FString& OutReason) const
100{
101 if (!OwnerDasan)
102 {
103 OutReason = TEXT("다산을 찾을 수 없습니다.");
104 return false;
105 }
106
107 // 이미 Answer 상태이고, 다른 플레이어가 질문 중이면 거부
108 if (OwnerDasan->DasanState == EDasanState::Answer && !CurQuestionerName.IsEmpty())
109 {
110 if (CurQuestionerName != PlayerName)
111 {
112 OutReason = FString::Printf(TEXT("다산은 %s에게 답변중입니다."), *CurQuestionerName);
113 return false;
114 }
115 }
116
117 return true;
118}

120bool UAnswerStateSystem::TryStartAnswer(const FString& PlayerName)
121{
122 if (!OwnerDasan || !OwnerDasan->HasAuthority())
123 return false;
124
125 FString Reason;
126 if (!CanStartAnswer(PlayerName, Reason))
127 {
128 // 거부 메시지를 요청한 플레이어에게 표시
129 PRINTLOG(TEXT("[AnswerSystem] Answer 시작 거부: %s"), *Reason);
130
131 // 모든 클라이언트에 다이얼로그 표시 (Toast는 로컬 플레이어만 보임)
132 if (auto DM = UDialogManager::Get(GetWorld()))
133 DM->ShowToast(Reason);
134
135 return false;
136 }
137
138 // Answer 시작 성공
139 CurQuestionerName = PlayerName;
140 PRINTLOG(TEXT("[AnswerSystem] %s님의 질문 시작"), *PlayerName);
141
142 // 모든 클라이언트에 Toast 메시지 표시
143 if (auto GS = GetWorld()->GetGameState<AYisanGameState>())
144 {
145 GS->MulticastRPC_ToastMessage(FString::Printf(TEXT("%s가 질문중입니다"), *PlayerName));
146 }
147
148 // 타임아웃 타이머 시작 (예외 상황 대비)
149 if (GetWorld())
150 {
151 GetWorld()->GetTimerManager().SetTimer(
152 AnswerTimeoutTimer,
153 this,
154 &UAnswerStateSystem::OnAnswerTimeout,
155 AnswerTimeoutDuration,
156 false
157 );
158 PRINTLOG(TEXT("[AnswerSystem] 답변 타임아웃 타이머 시작 (%.1f초)"), AnswerTimeoutDuration);
159 }
160
161 // 현재 Dasan 메인 상태가 Answer가 아닐 때만 처리
162 if (OwnerDasan->DasanState != EDasanState::Answer)
163 {
164 // 이전 상태 저장
165 PreviousMainState = OwnerDasan->DasanState;
166 PRINTLOG(TEXT("[AnswerSystem] 이전 메인 상태 저장: %s"),
167 *ENUM_TO_NAME(EDasanState, PreviousMainState));
168
169 // AI 이동 중지
170 if (OwnerDasan->DasanAicontrol)
171 {
172 OwnerDasan->DasanAicontrol->StopMovement();
173 PRINTLOG(TEXT("[AnswerSystem] AI 이동 중지"));
174 }
175
176 // Answer 상태로 전환
177 OwnerDasan->TransitionToState(EDasanState::Answer);
178 }
179
180 // Answer 서브 상태를 AnswerListen으로 설정 (음성 듣는 중)
181 SetCurState(EAnswerState::AnswerListen);
182
183 return true;
184}

194void UAnswerStateSystem::FinishAnswer()
195{
196 if (!OwnerDasan || !OwnerDasan->HasAuthority())
197 return;
198
199 PRINTLOG(TEXT("[AnswerSystem] %s님의 질문 종료"), *CurQuestionerName);
200
201 // 타임아웃 타이머 정리
202 if (GetWorld() && AnswerTimeoutTimer.IsValid())
203 {
204 GetWorld()->GetTimerManager().ClearTimer(AnswerTimeoutTimer);
205 PRINTLOG(TEXT("[AnswerSystem] 답변 타임아웃 타이머 정리"));
206 }
207
208 CurQuestionerName.Empty();
209
210 // Answer 서브 상태 초기화
211 SetCurState(EAnswerState::AnswerListen);
212
213 // 이전 상태로 복귀
214 if (PreviousMainState != EDasanState::Answer)
215 {
216 PRINTLOG(TEXT("[AnswerSystem] 이전 상태로 복귀: %s"), *ENUM_TO_NAME(EDasanState, PreviousMainState));
217 OwnerDasan->TransitionToState(PreviousMainState);
218 }
219}

221void UAnswerStateSystem::OnAnswerTimeout()
222{
223 if (!OwnerDasan || !OwnerDasan->HasAuthority())
224 return;
225
226 PRINTLOG(TEXT("[AnswerSystem] 답변 타임아웃 발생 - 강제 종료"));
227
228 // 타임아웃 메시지 표시
229 if (auto DM = UDialogManager::Get(GetWorld()))
230 DM->ShowToast(TEXT("질문 시간이 초과되어 대화가 종료되었습니다."));
231
232 // 답변 종료 처리
233 FinishAnswer();
234}
