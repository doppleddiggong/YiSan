20void USmallPopup::UpdateBuildingInfo(const EBuildingType InBuildingType)
21{
22 if (BuildingType == InBuildingType)
23 return;
24
25 // 데이터 매니저 가져오기
26 auto DataManager = UGameDataManager::Get(GetWorld());
27 if (!DataManager)
28 return;
29
30 // 건물 테이블 정보 데이터 가져오기
31 FBuildingData BuildingData;
32 if (DataManager->GetBuildingData(InBuildingType, BuildingData))
33 {
34 TitleText->SetText(FText::FromString(BuildingData.detailtitle));
35 DescText->SetText(FText::FromString(BuildingData.description));
36 }
37
38 // 건물 에셋 정보 데이터 가져오기
39 FBuildingAssetData AssetData;
40 if ( DataManager->GetBuildingAssetData(InBuildingType, AssetData) )
41 {
42 UBuildingDetailData* DetailAsset = AssetData.BuildingDetailDataAsset.LoadSynchronous();
43
44 if ( DetailAsset )
45 {
46 TSoftObjectPtr<UTexture2D> LoadedTexture;
47 if ( DetailAsset->LoadThumbnailImage(LoadedTexture) && LoadedTexture.Get() )
48 ThumbnailImage->SetBrushFromTexture(LoadedTexture.Get());
49 }
50 }
51}
