68void UGameDataManager::LoadData_BuildingDataTable()
69{
70 BuildingDataCache.Reset();
71 bLoadBuildingData = false;
72
73 UDataTable* TableObj = BuildingDataTable.LoadSynchronous();
74 if (!TableObj)
75 {
76 PRINTLOG(TEXT("Load failed: %s"), *BuildingDataTable.ToString());
77 return;
78 }
79
80 static const FString ContextString(TEXT("BuildingDataTable"));
81 for (const FName& RowName : TableObj->GetRowNames() )
82 {
83 if (const FBuildingData* Row = TableObj->FindRow<FBuildingData>(RowName, ContextString, true))
84 {
85 BuildingDataCache.Add(Row->Type, *Row);
86 }
87 }
88
89 bLoadBuildingData = true;
90}
