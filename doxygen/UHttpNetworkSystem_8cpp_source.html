70void UHttpNetworkSystem::RequestHealth( FResponseHealthDelegate InDelegate )
71{
72 auto HttpRequest = FHttpModule::Get().CreateRequest();
73
74 HttpRequest->SetVerb(NETWORK_GET);
75 HttpRequest->SetURL(NetworkConfig::GetFullUrl(RequestAPI::Health));
76 HttpRequest->SetHeader(TEXT("Content-Type"), TEXT("application/json"));
77
78 LogNetwork(ENetworkLogType::Get, *HttpRequest->GetURL());
79
80 HttpRequest->OnProcessRequestComplete().BindLambda(
81 [WeakThis = TWeakObjectPtr<UHttpNetworkSystem>(this), InDelegate](FHttpRequestPtr Req, FHttpResponsePtr ResPtr, bool bWasSuccessful)
82 {
83 if (!WeakThis.IsValid() || IsEngineExitRequested())
84 return;
85
86 WeakThis->AddNetworkWaitCount(-1);
87 FResponseHealth ResponseData;
88 ResponseData.SetFromHttpResponse(ResPtr);
89
90 InDelegate.ExecuteIfBound(ResponseData, bWasSuccessful);
91 });
92
93 AddNetworkWaitCount(1);
94 HttpRequest->ProcessRequest();
95}

97void UHttpNetworkSystem::RequestASK(const FString& FilePath, const FGPTContext& Context, FResponseAskDelegate InDelegate)
98{
99 auto HttpRequest = FHttpModule::Get().CreateRequest();
100 HttpRequest->SetVerb(NETWORK_POST);
101 HttpRequest->SetURL(NetworkConfig::GetFullUrl(RequestAPI::ASK));
102 HttpRequest->SetHeader(TEXT("Accept"), TEXT("application/json"));
103
104 FHttpMultipartFormData Form;
105 if (!Form.AddFile(TEXT("file"), FilePath))
106 {
107 NETWORK_LOG(TEXT("Ask: file load failed: %s"), *FilePath);
108 return;
109 }
110 Form.SetupHttpRequest(HttpRequest);
111
112 FRequestASK RequestData;
113 RequestData.context = Context;
114
115 FString RequestBody;
116 if (!RequestData.ToJsonString(RequestBody))
117 {
118 NETWORK_LOG(TEXT("Failed to serialize FRequestASK to JSON"));
119 return;
120 }
121
122 Form.AddStringField(TEXT("context"), RequestBody);
123 Form.SetupHttpRequest(HttpRequest);
124
125 const FString LogBody = FString::Printf(TEXT("file: %s, context: %s"), *FilePath, *RequestBody);
126 LogNetwork(ENetworkLogType::Post, *HttpRequest->GetURL(), LogBody);
127
128 HttpRequest->OnProcessRequestComplete().BindLambda(
129 [WeakThis = TWeakObjectPtr<UHttpNetworkSystem>(this), InDelegate](FHttpRequestPtr Req, FHttpResponsePtr ResPtr, bool bWasSuccessful)
130 {
131 if (!WeakThis.IsValid() || IsEngineExitRequested())
132 return;
133
134 WeakThis->AddNetworkWaitCount(-1);
135 FResponseAsk ResponseData;
136 if (bWasSuccessful && ResPtr.IsValid())
137 {
138 NETWORK_LOG(TEXT("[RES] Ask completed: transcribed_text, gpt_response_text, audio_content"));
139 ResponseData.SetFromHttpResponse(ResPtr);
140 }
141 InDelegate.ExecuteIfBound(ResponseData, bWasSuccessful);
142 });
143
144 AddNetworkWaitCount(1);
145 HttpRequest->ProcessRequest();
146}
