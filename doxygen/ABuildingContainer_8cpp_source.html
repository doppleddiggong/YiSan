49void ABuildingContainer::RefreshBuildingMap()
50{
51 UWorld* World = GetWorld();
52 if (!World)
53 {
54 return;
55 }
56
57 BuildingMap.Reset();
58
59 for (TActorIterator<ABuilding> It(World); It; ++It)
60 {
61 ABuilding* const Building = *It;
62 if (!IsValid(Building))
63 continue;
64
65 const EBuildingType BuildingType = Building->BuildingType;
66 if (BuildingType == EBuildingType::None)
67 continue;
68
69 if (const TObjectPtr<ABuilding>* const Existing = BuildingMap.Find(BuildingType))
70 PRINTLOG( TEXT("Duplicate building type %s found"), *FString(ENUM_TO_NAME(EBuildingType, BuildingType) ));
71
72 BuildingMap.Add(BuildingType, Building);
73 }
74}
