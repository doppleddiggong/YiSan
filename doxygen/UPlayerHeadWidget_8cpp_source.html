30void UPlayerHeadWidget::CheckPlayerState()
31{
32 // 유효한 텍스트 블록과 소유 액터가 있을 경우에만 로직 실행
33 if (OwningActor.IsValid())
34 {
35 // Pawn에서 PlayerState를 가져와 AYiSanPlayerState로 캐스팅합니다.
36 if (auto PS = OwningActor->GetPlayerState<AYiSanPlayerState>())
37 {
38 // 닉네임이 비어있지 않다면 텍스트를 설정합니다.
39 if (!PS->Nickname.IsEmpty())
40 {
41 // 이름이 설정되었으므로, 타이머를 정지시킵니다.
42 GetWorld()->GetTimerManager().ClearTimer(TimerHandle_CheckPlayerState);
43
44 PlayerNameText->SetText(FText::FromString(PS->Nickname));
45
46 // 로컬 플레이어인 경우에만 녹색으로 설정
47 if (OwningActor->IsLocallyControlled())
48 {
49 PlayerNameText->SetColorAndOpacity(FLinearColor::Green);
50 }
51 else
52 {
53 PlayerNameText->SetColorAndOpacity(FLinearColor::White);
54 }
55
56 image_Icon->SetBrushFromTexture( LoadObject<UTexture2D>(nullptr, *PS->GetResourcePath()));
57 OwningActor->ChatPlayerSystem->AnnouncePlayerJoin();
58 }
59 }
