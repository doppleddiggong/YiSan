36void UMegaPopup::UpdateBuildingInfo(const EBuildingType InBuildingType)
37{
38 if (BuildingType == InBuildingType)
39 return;
40
41 // 데이터 매니저 가져오기
42 auto DataManager = UGameDataManager::Get(GetWorld());
43 if (!DataManager)
44 return;
45
46 // 건물 테이블 정보 데이터 가져오기
47 FBuildingData BuildingData;
48 if (DataManager->GetBuildingData(InBuildingType, BuildingData))
49 {
50 DetailTitleText->SetText(FText::FromString(BuildingData.detailtitle));
51 DetailText->SetText(FText::FromString(BuildingData.detaildesc));
52 SubTitle->SetText(FText::FromString(BuildingData.subtile));
53 }
54
55 // 건물 에셋 정보 데이터 가져오기
56 FBuildingAssetData AssetData;
57 if (DataManager->GetBuildingAssetData(InBuildingType, AssetData))
58 {
59 UBuildingDetailData* DetailAsset = AssetData.BuildingDetailDataAsset.LoadSynchronous();
60
61 if (DetailAsset)
62 {
63 BuildDetailAsset = DetailAsset;
64
65 const bool bUseMedia = DetailAsset->UseMedia;
66 const bool bPlayedMedia = bUseMedia && PlayMedia(*DetailAsset);
67 if (!bUseMedia || !bPlayedMedia)
68 {
69 DisplayBackgroundImage(*DetailAsset);
70 }
71
72 TSoftObjectPtr<USoundCue> LoadedCue;
73 if (DetailAsset->LoadSoundCue(LoadedCue) && LoadedCue.Get())
74 {
75 if (auto SoundManager = UGameSoundManager::Get(GetWorld()))
76 SoundManager->PlayConversationVoice(LoadedCue.Get());
77 }
78 }
79 }
80
81 BuildingType = InBuildingType;
82}

102bool UMegaPopup::PlayMedia(const UBuildingDetailData& DetailAsset)
103{
104 if (!MediaPlayer)
105 {
106 PRINTLOG(TEXT("UMegaPopup::PlayBackgroundMedia - MediaPlayer is not assigned."));
107 return false;
108 }
109
110 if (!BackgroundMedia)
111 {
112 PRINTLOG(TEXT("UMegaPopup::PlayBackgroundMedia - MediaImage is not assigned."));
113 return false;
114 }
115
116 TSoftObjectPtr<UMediaSource> LoadedMediaSource;
117 if (!DetailAsset.LoadBackgroundVideoSource(LoadedMediaSource) || !LoadedMediaSource.Get())
118 {
119 PRINTLOG(TEXT("UMegaPopup::PlayBackgroundMedia - Failed to load media source."));
120 return false;
121 }
122
123 StopMedia();
124
125 bPendingMedia = true;
126 BuildDetailAsset = &DetailAsset;
127
128 MediaPlayer->OnMediaOpened.AddDynamic(this, &UMegaPopup::OnMediaOpened);
129 MediaPlayer->OnMediaOpenFailed.AddDynamic(this, &UMegaPopup::OnMediaOpenFailed);
130
131 if (!MediaPlayer->OpenSource(LoadedMediaSource.Get()))
132 {
133 PRINTLOG(TEXT("UMegaPopup::PlayBackgroundMedia - Failed to open media source."));
134 ClearMediaDelegates();
135 bPendingMedia = false;
136 return false;
137 }
138
139 BackgroundMedia->SetVisibility(ESlateVisibility::Collapsed);
140
141 return true;
142}

171void UMegaPopup::OnMediaOpened(FString OpenedUrl)
172{
173 if (!MediaPlayer || !bPendingMedia)
174 return;
175
176 ClearMediaDelegates();
177
178 bPendingMedia = false;
179
180 if (!MediaPlayer->Play())
181 {
182 PRINTLOG(TEXT("UMegaPopup::HandleMediaOpened - Failed to start playback after open."));
183
184 if (BuildDetailAsset)
185 DisplayBackgroundImage(*BuildDetailAsset);
186 return;
187 }
188
189 if (BackgroundMedia)
190 BackgroundMedia->SetVisibility(ESlateVisibility::Visible);
191
192 if (BackgroundImage)
193 BackgroundImage->SetVisibility(ESlateVisibility::Collapsed);
194}

203
204 if (BuildDetailAsset)

208void UMegaPopup::DisplayBackgroundImage(const UBuildingDetailData& DetailAsset)
209{
210 StopMedia();
211
212 if (!BackgroundImage)
213 return;
214
215 TSoftObjectPtr<UTexture2D> LoadedTexture;
216 if (DetailAsset.LoadBackgroundImage(LoadedTexture) && LoadedTexture.Get())
217 BackgroundImage->SetBrushFromTexture(LoadedTexture.Get());
218
219 BackgroundImage->SetVisibility(ESlateVisibility::Visible);
220
221 if (BackgroundMedia)
222 BackgroundMedia->SetVisibility(ESlateVisibility::Collapsed);
223}
