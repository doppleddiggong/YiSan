71void APlayerControl::BeginPlay()
72{
73 Super::BeginPlay();
74
75 if (IsLocalController())
76 {
77 if (auto LP = GetLocalPlayer())
78 {
79 if (auto SubSystem = LP->GetSubsystem<UEnhancedInputLocalPlayerSubsystem>())
80 {
81 if (IMC_Default)
82 {
83 SubSystem->ClearAllMappings();
84 SubSystem->AddMappingContext(IMC_Default, 0);
85 }
86 }
87 }
88
89 // 클라이언트에서 닉네임을 서버로 전송
90 if (UYiSanGameInstance* GI = GetGameInstance<UYiSanGameInstance>())
91 {
92 if (UYisanOnlineSystem* NetworkSubsystem = GI->GetSubsystem<UYisanOnlineSystem>())
93 {
94 FString Nickname = NetworkSubsystem->GetPlayerNickname();
95 if (!Nickname.IsEmpty())
96 {
97 ServerRPC_SetPlayerNickname(Nickname);
98 }
99 }
100 }
101 }
102
103 // 한 프레임 뒤 보정 체크
104 GetWorldTimerManager().SetTimerForNextTick([this]()
105 {
106 if (!GetPawn())
107 {
108 PRINTLOG( TEXT("[YiSan_PC] Pawn missing - request restart"));
109 if (AGameModeBase* GM = UGameplayStatics::GetGameMode(this))
110 {
111 GM->RestartPlayer(this); // GameMode 통해 재스폰
112 }
113 }
114 else
115 {
116 PRINTLOG( TEXT("[YiSan_PC] Pawn OK: %s"), *GetNameSafe(GetPawn()));
117 GetPawn()->EnableInput(this);
118 }
119 });
120
121
122 if ( auto BroadcastManager = UBroadcastManager::Get(GetWorld()) )
123 BroadcastManager->OnPlayerControlState.AddDynamic(this, &APlayerControl::OnPlayerControlState);
124}

233void APlayerControl::OnPlayerControlState(bool bState, UUserWidget* FocusWidget)
234{

236 {
237 // 플레이어 컨트롤 Enable 한다
238 FInputModeGameOnly InputMode;

240 this->SetInputMode(InputMode);
241 this->SetShowMouseCursor(false);
242
243 GetPawn()->EnableInput(this);
244 }
245 else
246 {
247 // 플레이어 컨트롤 disable 하자
248 FInputModeUIOnly InputMode;
249 if (IsValid(FocusWidget))

252 this->SetInputMode(InputMode);
253 this->SetShowMouseCursor(true);

377void APlayerControl::ServerStartMapTravel(const FString& MapPath)
378{
379 // 서버 권한 체크
380 if (!HasAuthority())
381 {
382 PRINTLOG( TEXT("[Travel] 클라이언트가 ServerStartMapTravel 호출 시도 - 무시됨"));
383 return;
384 }
385
386 PRINTLOG( TEXT("[Travel] 서버가 맵 전환 시작: %s"), *MapPath);
387
388 // 약간의 딜레이 후 ServerTravel 호출 (UI가 먼저 뜨도록)
389 FTimerHandle TravelTimerHandle;
390 GetWorldTimerManager().SetTimer(TravelTimerHandle, [this, MapPath]()
391 {
392 if (UWorld* World = GetWorld())
393 {
394 for (FConstPlayerControllerIterator It = World->GetPlayerControllerIterator(); It; ++It)
395 {
396 if (APlayerControl* PlayerControl = Cast<APlayerControl>(It->Get()))
397 {
398 if (PlayerControl->IsLocalController())
399 {
400 PlayerControl->ShowLoadingScreenLocal();
401 }
402 else
403 {
404 PlayerControl->ClientRPC_ShowLoadingTransition();
405 }
406 }
407 }
408 }
409
410 if (auto TM = ULoadingTransitionManager::Get(this))
411 {
412 TM->ShowLoadingScreen();
413 }
414
415 if (auto GI = UYiSanLoading::Get(GetWorld()))
416 {
417 FString TravelURL = MapPath + TEXT("?listen");
418 GI->InitSystem(TravelURL, false, true);
419 }
420 }, 0.1f, false);
421}

435void APlayerControl::HandleLoadingComplete()
436{
437 PRINTLOG(TEXT("[LOADING_FLOW] HandleLoadingComplete called - IsLocal: %s"), IsLocalController() ? TEXT("true") : TEXT("false"));
438
439 if (!IsLocalController())
440 return;
441
442 // 게스트가 join한 경우 PostLoadMapWithWorld가 호출되지 않을 수 있음
443 // 이 경우 로딩 상태를 강제로 완료 처리
444 if (auto LoadingSubsystem = UYiSanLoading::Get(GetWorld()))
445 {
446 if (!LoadingSubsystem->IsLoadingComplete())
447 {
448 PRINTLOG(TEXT("[LOADING_FLOW] Loading not complete - attempting to force complete for guest"));
449 LoadingSubsystem->ForceCompleteForGuest();
450 }
451 }
452
453 if (!IsReadyToFinish())
454 {
455 if (!bAwaitFinish)
456 {
457 if (auto CtrlPawn = GetPawn())
458 PRINTLOG(TEXT("[LOADING_FLOW] Pawn not fully initialized - deferring loading completion for %s"), *GetNameSafe(CtrlPawn));
459 else
460 PRINTLOG(TEXT("[LOADING_FLOW] Pawn not ready - deferring loading completion for %s"), *GetName());
461 }
462
463 bAwaitFinish = true;
464 return;
465 }
466
467 PRINTLOG(TEXT("[LOADING_FLOW] All conditions met - proceeding to CompleteLoading"));
468 CompleteLoading();
469}

495 }
496
497 SetInputMode(FInputModeGameOnly());

500 PRINTLOG(TEXT("[Travel] Loading transition completed for %s"), *GetName());
501 },
502 3.0f, // 렌더링 준비 대기 시간
503 false);

540bool APlayerControl::IsReadyToFinish() const
541{
542 PRINTLOG(TEXT("[LOADING_FLOW] IsReadyToFinish check - Pawn: %s, bPawnReady: %s, bPawnHasName: %s"),
543 GetPawn() ? TEXT("true") : TEXT("false"),
544 bPawnReady ? TEXT("true") : TEXT("false"),
545 bPawnHasName ? TEXT("true") : TEXT("false"));
546
547 // Pawn과 Name 준비 체크
548 if (GetPawn() == nullptr)
549 {
550 PRINTLOG(TEXT("[LOADING_FLOW] NOT READY: Pawn is null"));
551 return false;
552 }
553
554 if (!bPawnReady)
555 {
556 PRINTLOG(TEXT("[LOADING_FLOW] NOT READY: bPawnReady is false"));
557 return false;
558 }
559
560 if (!bPawnHasName)
561 {
562 PRINTLOG(TEXT("[LOADING_FLOW] NOT READY: bPawnHasName is false"));
563 return false;
564 }
565
566 // 클라이언트 측 로딩 완료 체크 (WorldPartition, Texture, LevelInstance)
567 if (auto LoadingSubsystem = UYiSanLoading::Get(GetWorld()))
568 {
569 bool bLoadingComplete = LoadingSubsystem->IsLoadingComplete();
570 PRINTLOG(TEXT("[LOADING_FLOW] Client-side loading complete: %s"), bLoadingComplete ? TEXT("true") : TEXT("false"));
571
572 if (!bLoadingComplete)
573 {
574 PRINTLOG(TEXT("[LOADING_FLOW] NOT READY: Client-side loading not complete yet - deferring"));
575 return false;
576 }
577 }
578 else
579 {
580 PRINTLOG(TEXT("[LOADING_FLOW] WARNING: LoadingSubsystem not found!"));
581 }
582
583 PRINTLOG(TEXT("[LOADING_FLOW] READY: All conditions satisfied!"));
584 return true;
585}
