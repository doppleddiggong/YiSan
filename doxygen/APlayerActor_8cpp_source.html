205void APlayerActor::Cmd_Move_Implementation(const FVector2D& Axis)
206{
207 if ( !Controller)
208 return;
209
210 const FRotator ControlRotation = Controller->GetControlRotation();
211 const FRotator YawRotation(0.0f, ControlRotation.Yaw, 0.0f);
212
213 const FVector ForwardDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::X);
214 const FVector RightDirection = FRotationMatrix(YawRotation).GetUnitAxis(EAxis::Y);
215
216 AddMovementInput(ForwardDirection, Axis.Y);
217 AddMovementInput(RightDirection, Axis.X);
218}

307void APlayerActor::OnExecVoiceCommand(EVoiceCommandType InType, AActor* Requester)
308{
309 PRINT_STRING(TEXT("%s from %s"),
310 *FString( ENUM_TO_NAME(EVoiceCommandType, InType)),
311 Requester ? *Requester->GetName() : TEXT("Unknown"));
312
313 // 요청자가 플레이어인지 확인
314 APlayerActor* RequesterPlayer = Cast<APlayerActor>(Requester);
315 if (!RequesterPlayer)
316 return;
317
318 // Cmd_BuildingInfo는 로컬 클라이언트에서만 실행 (UI 표시용)
319 if (InType == EVoiceCommandType::Cmd_BuildingInfo)
320 {
321 // 요청자 본인만 실행
322 if (this != RequesterPlayer || !IsLocallyControlled())
323 return;
324
325 // 가장 가까운 건물 찾기
326 auto AllBuildings = FComponentHelper::GetAllOfClass<ABuilding>(GetWorld());
327 if (AllBuildings.Num() == 0)
328 {
329 PRINTLOG(TEXT("[VoiceCommand] 주변에 건물이 없습니다"));
330 return;
331 }
332
333 float MinDistance = TNumericLimits<float>::Max();
334 ABuilding* NearestBuilding = nullptr;
335
336 FVector MyLocation = GetActorLocation();
337 for (auto Building : AllBuildings)
338 {
339 float Distance = FVector::Dist(MyLocation, Building->GetActorLocation());
340 if (Distance < MinDistance)
341 {
342 MinDistance = Distance;
343 NearestBuilding = Building;
344 }
345 }
346
347 if (NearestBuilding && MainWidgetInst)
348 {
349 MainWidgetInst->ShowMegaPopup(NearestBuilding->BuildingType);
350
351 PRINTLOG(TEXT("[VoiceCommand] %s 정보 표시"),
352 *UGameDataManager::Get(GetWorld())->GetBuildingDataName(NearestBuilding->BuildingType));
353 }
354 return;
355 }
356
357 // 나머지 명령들은 서버에서만 실행
358 if (!HasAuthority())
359 return;
360
361 if (InType == EVoiceCommandType::Cmd_Call)
362 {
363 // 다른 모든 플레이어를 요청자에게 집결
364 // 요청자 본인은 제외
365 if (this == RequesterPlayer)
366 return;
367
368 FVector RequesterLocation = RequesterPlayer->GetActorLocation();
369 FRotator RequesterRotation = RequesterPlayer->GetActorRotation();
370
371 // 요청자의 앞쪽으로 200 유닛 떨어진 위치 계산
372 FVector TargetLocation = RequesterLocation + (RequesterRotation.Vector() * 200.f);
373
374 // 자신을 요청자 앞으로 텔레포트
375 TeleportTo(TargetLocation, GetActorRotation(), false, true);
376
377 PRINTLOG(TEXT("[VoiceCommand] %s가 %s에게 집결"), *GetPlayerDisplayName(), *RequesterPlayer->GetPlayerDisplayName());
378 }
379 else if (InType == EVoiceCommandType::Cmd_Friend)
380 {
381 // 요청자가 다른 플레이어(다음 번째 플레이어)에게로 이동
382 // 요청자만 실행
383 if (this != RequesterPlayer)
384 return;
385
386 // 모든 플레이어 가져오기
387 auto AllPlayers = FComponentHelper::GetAllOfClass<APlayerActor>(GetWorld());
388 if (AllPlayers.Num() <= 1)
389 {
390 PRINTLOG(TEXT("[VoiceCommand] 이동할 다른 플레이어가 없습니다"));
391 return;
392 }
393
394 // 자신의 PlayerIndex 가져오기
395 int32 MyIndex = GetPlayerIndex();
396
397 // 다음 플레이어 찾기 (순환)
398 APlayerActor* TargetPlayer = nullptr;
399 for (auto Player : AllPlayers)
400 {
401 int32 PlayerIdx = Player->GetPlayerIndex();
402 if (PlayerIdx > MyIndex)
403 {
404 TargetPlayer = Player;
405 break;
406 }
407 }
408
409 // 다음 플레이어가 없으면 첫 번째 플레이어로
410 if (!TargetPlayer)
411 {
412 for (auto Player : AllPlayers)
413 {
414 if (Player != this)
415 {
416 TargetPlayer = Player;
417 break;
418 }
419 }
420 }
421
422 if (TargetPlayer)
423 {
424 FVector TargetLocation = TargetPlayer->GetActorLocation();
425 FRotator TargetRotation = TargetPlayer->GetActorRotation();
426
427 // 대상 플레이어의 앞쪽으로 200 유닛 떨어진 위치 계산
428 FVector TeleportLocation = TargetLocation + (TargetRotation.Vector() * 200.f);
429
430 // 텔레포트
431 TeleportTo(TeleportLocation, GetActorRotation(), false, true);
432
433 PRINTLOG(TEXT("[VoiceCommand] %s가 %s에게로 이동"), *GetPlayerDisplayName(), *TargetPlayer->GetPlayerDisplayName());
434 }
435 }
436 else if (InType == EVoiceCommandType::Cmd_Target)
437 {
438 // 요청자가 현재 퀘스트 목표 위치로 이동
439 // 요청자만 실행
440 if (this != RequesterPlayer)
441 return;
442
443 // 퀘스트 매니저에서 현재 목표 가져오기
444 AQuestManagerActor* QuestManager = AQuestManagerActor::Get(this);
445 if (!QuestManager || !QuestManager->HasActiveQuest())
446 {
447 PRINTLOG(TEXT("[VoiceCommand] 활성화된 퀘스트가 없습니다"));
448 return;
449 }
450
451 EBuildingType TargetBuildingType = QuestManager->GetCurrentTarget();
452
453 // 목표 건물 찾기
454 auto AllBuildings = FComponentHelper::GetAllOfClass<ABuilding>(GetWorld());
455 ABuilding* TargetBuilding = nullptr;
456 for (auto Building : AllBuildings)
457 {
458 if (Building->BuildingType == TargetBuildingType)
459 {
460 TargetBuilding = Building;
461 break;
462 }
463 }
464
465 if (TargetBuilding)
466 {
467 FVector BuildingLocation = TargetBuilding->GetActorLocation();
468 FRotator BuildingRotation = TargetBuilding->GetActorRotation();
469
470 // 건물 앞쪽으로 200 유닛 떨어진 위치 계산
471 FVector TeleportLocation = BuildingLocation + (BuildingRotation.Vector() * 200.f);
472
473 // 텔레포트
474 TeleportTo(TeleportLocation, GetActorRotation(), false, true);
475
476 PRINTLOG(TEXT("[VoiceCommand] %s가 퀘스트 목표 %s로 이동"),
477 *GetPlayerDisplayName(),
478 *UGameDataManager::Get(GetWorld())->GetBuildingDataName(TargetBuildingType));
479 }
480 else
481 {
482 PRINTLOG(TEXT("[VoiceCommand] 퀘스트 목표 건물을 찾을 수 없습니다"));
483 }
484 }
485}
