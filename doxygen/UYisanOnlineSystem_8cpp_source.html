30void UYisanOnlineSystem::Initialize(FSubsystemCollectionBase& Collection)
31{
32 Super::Initialize(Collection);
33 if( auto subsys = Online::GetSubsystem(GetWorld()) )
34 {
35 //서브시스템의 인터페이스를 가져오자
36 sessionInterface = subsys->GetSessionInterface();
37 //세션생성 성공시 호출되는 함수 등록
38 sessionInterface->OnCreateSessionCompleteDelegates.AddUObject(this, &UYisanOnlineSystem::OnCreateSessionComplete);
39 //세션조회 성공시 호출되는 함수 등록
40 sessionInterface->OnFindSessionsCompleteDelegates.AddUObject(this, &UYisanOnlineSystem::OnFindSessionComplete);
41 //세션참여 성공시 호출되는 함수 등록
42 sessionInterface->OnJoinSessionCompleteDelegates.AddUObject(this, &UYisanOnlineSystem::OnJoinSessionComplete);
43 }
44}

61void UYisanOnlineSystem::RequestRefreshPlayerList()
62{
63 if (CachedGameState)
64 {
65 TArray<FString> PlayerList = CachedGameState->GetPlayerList();
66 if (PlayerList.Num() > 0)
67 OnPlayerListUpdated.Broadcast(PlayerList);
68 else
69 CachedGameState->RequestRefreshPlayerList();
70 return;
71 }
72
73 PRINTLOG(TEXT("RequestPlayerListRefresh Failed!"));
74
75 if (UWorld* World = GetWorld())
76 {
77 FTimerHandle RetryTimerHandle;
78 World->GetTimerManager().SetTimer(RetryTimerHandle, [this]()
79 {
80 if (CachedGameState)
81 {
82 PRINTLOG(TEXT("RequestRefreshPlayerList - Retry Succeeded"));
83 auto PlayerList = CachedGameState->GetPlayerList();
84 if (PlayerList.Num() > 0)
85 OnPlayerListUpdated.Broadcast(PlayerList);
86 else
87 CachedGameState->RequestRefreshPlayerList();
88 }
89 else
90 {
91 PRINTLOG(TEXT("RequestPlayerListRefresh - Still Failed"));
92 }
93 }, 0.5f, false);
94 }
95}

104void UYisanOnlineSystem::CreateMySession(FString displayName, int32 playerCount)
105{
106 if (!sessionInterface.IsValid())
107 {
108 PRINTLOG( TEXT("SessionInterface is invalid!"));
109 return;
110 }
111
112 FOnlineSessionSettings sessionSettings;
113 FName subsysName = Online::GetSubsystem(GetWorld())->GetSubsystemName();
114 sessionSettings.bIsLANMatch = subsysName.IsEqual(FName(TEXT("NULL")));
115 sessionSettings.bUseLobbiesIfAvailable = true;
116 sessionSettings.bUsesPresence = true;
117 sessionSettings.bShouldAdvertise = true;
118 sessionSettings.NumPublicConnections = playerCount;
119 sessionSettings.Set(FName("DP_NAME"), displayName, EOnlineDataAdvertisementType::ViaOnlineServiceAndPing);
120
121 FUniqueNetIdPtr netID = GetWorld()->GetFirstLocalPlayerFromController()->GetUniqueNetIdForPlatformUser().GetUniqueNetId();
122 FString netIDString = netID->ToString();
123 PRINTLOG( TEXT("서브시스템 : %s"), *subsysName.ToString());
124
125 sessionInterface->CreateSession(0, FName(displayName), sessionSettings);
126}

148void UYisanOnlineSystem::FindOtherSession()
149{
150 PRINTLOG( TEXT("FindOtherSession()"));
151
152 if (auto DM = UDialogManager::Get(GetWorld()))
153 DM->ShowToast(TEXT("조회 시작"));
154
155 //sessionSearch 만들자
156 sessionSearch = MakeShared<FOnlineSessionSearch>();
157 FName subsysName = Online::GetSubsystem(GetWorld())->GetSubsystemName();
158 sessionSearch->bIsLanQuery = subsysName.IsEqual(FName(TEXT("NULL")));
159 sessionSearch->QuerySettings.Set(SEARCH_LOBBIES, true, EOnlineComparisonOp::Equals);
160 sessionSearch->MaxSearchResults = 100;
161 sessionInterface->FindSessions(0, sessionSearch.ToSharedRef());
162}

165void UYisanOnlineSystem::OnFindSessionComplete(bool success)
166{
167 PRINTLOG( TEXT("OnFindSessionComplete(%d)"), success );
168
169 if (UDialogManager* DM = UDialogManager::Get(GetWorld()))
170 DM->ShowToast(TEXT("조회 완료"));
171
172 if (success)
173 {
174 if (sessionSearch.IsValid())
175 { // Check if sessionSearch is valid
176 auto results = sessionSearch->SearchResults;
177 for (int32 i = 0; i < results.Num(); i++)
178 {
179 //세션이름담을변수
180 FString displayName;
181 results[i].Session.SessionSettings.Get(FName(TEXT("DP_NAME")), displayName);
182
183 PRINTLOG( TEXT("세션 : %i, 이름 : %s"), i, *displayName);
184 OnFindComplete.ExecuteIfBound(i, displayName);
185 }
186 }
187 }
188
189 OnFindComplete.ExecuteIfBound(-1, FString());
190}

193void UYisanOnlineSystem::JoinOtherSession(int32 sessionIndex)
194{
195 auto results = sessionSearch->SearchResults;
196 if (results.Num() <= 0)
197 {
198 PRINTLOG( TEXT("No sessions found! Cannot join."));
199 return;
200 }
201 results[sessionIndex].Session.SessionSettings.bUseLobbiesIfAvailable = true;
202 results[sessionIndex].Session.SessionSettings.bUsesPresence = true;
203
204 FString displayName;
205 results[sessionIndex].Session.SessionSettings.Get(FName(TEXT("DP_NAME")), displayName);
206
207 ULoadingCircleManager::Get(GetWorld())->Show();
208
209 sessionInterface->JoinSession(0, FName(displayName), results[sessionIndex]);
210}

213void UYisanOnlineSystem::OnJoinSessionComplete(FName sessionName, EOnJoinSessionCompleteResult::Type result)
214{
215 ULoadingCircleManager::Get(GetWorld())->Hide();
216
217 if (result == EOnJoinSessionCompleteResult::Success)
218 {
219 FString url;
220 sessionInterface->GetResolvedConnectString(sessionName, url);
221 PRINTLOG( TEXT("URL : %s"), *url);
222
223 if (auto Ctrl = GetWorld()->GetFirstPlayerController())
224 {
225 Ctrl->ClientTravel(url, TRAVEL_Absolute);
226 }
227 }
228 else
229 {
230 PRINTLOG( TEXT("OnJoinSessionComplete(%d)"), result);
231 }
232}
