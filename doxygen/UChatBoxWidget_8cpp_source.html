43void UChatBoxWidget::OnTextCommittedHandler(const FText& Text, ETextCommit::Type CommitMethod)
44{
45 if (CommitMethod != ETextCommit::OnEnter)
46 return;
47
48 const FString InputString = Text.ToString();
49
50 if (!InputString.IsEmpty())
51 {
52 if (ChatPlayerSystem)
53 {
54 int32 PlayerIndex = Owner->GetPlayerIndex();
55 FString PlayerName = Owner->GetPlayerDisplayName();
56
57 // 채팅 메시지 전송
58 FChatMessage ChatMessage(EChatMessageType::User, PlayerIndex, PlayerName, *InputString);
59 ChatPlayerSystem->ServerRPC_SendChatMessage(ChatMessage);
60
61 // GPT 질문 요청
62 this->Ask(InputString, Owner->GetGPTContext());
63 }
64 }
65
66 ExitChat();
67}

69void UChatBoxWidget::Ask(const FString& InMsg, const FGPTContext& SpatialContext)
70{
71 // 로컬 플레이어만 GPT 요청
72 if (!IsValid(Owner) || !Owner->IsLocallyControlled())
73 {
74 return;
75 }
76
77 if (auto ReqNetwork = UHttpNetworkSystem::Get(GetWorld()))
78 ReqNetwork->RequestGPT(InMsg, SpatialContext, FResponseAskDelegate::CreateUObject(this, &UChatBoxWidget::OnResponseAsk));
79}

81void UChatBoxWidget::OnResponseAsk(FResponseAsk& Response, bool bSuccess)
82{
83 if (bSuccess)
84 {
85 PRINTLOG(TEXT("OnResponseAsk: Received transcribed_text : %s"), *Response.transcribed_text);
86 PRINTLOG(TEXT("OnResponseAsk: Received gpt_response_text : %s"), *Response.gpt_response_text);
87 PRINTLOG(TEXT("OnResponseAsk: Received audio data size: %d"), Response.audio_data.Num());
88
89 auto VoiceCommand = UVoiceFunctionLibrary::GetVoiceCommand(Response.gpt_response_text);
90 PRINTLOG(TEXT("OnResponseAsk: VoiceCommand result is %d"), static_cast<int32>(VoiceCommand));
91
92 if ( VoiceCommand != EVoiceCommandType::None )
93 {
94 BroadcastManager->SendExecVoiceCommand( VoiceCommand, Owner );
95 }
96 else
97 {
98 FChatMessage ChatMessage(EChatMessageType::NPC, -1, GameString::NPC, Response.gpt_response_text);
99 ChatPlayerSystem->ServerRPC_SendChatMessage(ChatMessage);
100 }
101 }
102 else
103 {
104 PRINTLOG( TEXT("--- Network Response Received (FAIL) ---"));
105
106 // 실패한 경우에도 답변 종료 처리 (PlayerController를 통해 서버에 요청)
107 if (APlayerControl* PC = Owner->GetController<APlayerControl>())
108 {
109 PC->ServerRPC_FinishAnswer();
110 PRINTLOG(TEXT("[ChatBox] GPT 응답 실패 - FinishAnswer 호출"));
111 }
112 }
113}

149void UChatBoxWidget::Scroll(bool bUp)
150{
151 const float CurrentOffset = ScrollBox->GetScrollOffset();
152 const float Direction = bUp ? -1.0f : 1.0f;
153 float NewOffset = CurrentOffset + (ScrollMultiplier * Direction);
154
155 const float MaxOffset = ScrollBox->GetScrollOffsetOfEnd();
156 NewOffset = FMath::Clamp(NewOffset, 0.0f, MaxOffset);
157
158 ScrollBox->SetScrollOffset(NewOffset);
159}

161void UChatBoxWidget::AddChatMessage(const FChatMessage& ChatMessage)
162{
163 if (!ScrollBox || !ChatEntryClass)
164 return;
165
166 UChatEntryWidget* NewEntry = CreateWidget<UChatEntryWidget>(this, ChatEntryClass);
167 NewEntry->ChatMessageData = ChatMessage;

169 ChatMessagesBox->AddChildToVerticalBox(NewEntry);

172 NewEntry->InvalidateLayoutAndVolatility();
173 ChatMessagesBox->InvalidateLayoutAndVolatility();
174 ScrollBox->InvalidateLayoutAndVolatility();

177 FTimerHandle Handle;
178 GetWorld()->GetTimerManager().SetTimer(Handle, [this]()
179 {
180 ScrollBox->ScrollToEnd();
181 }, 0.1f, false);
