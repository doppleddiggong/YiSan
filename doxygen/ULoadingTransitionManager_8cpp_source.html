67void ULoadingTransitionManager::EnsureWidgetForWorld(UWorld* World)
68{
69 if (World == nullptr || !World->IsGameWorld())
70 return;
71
72 ULocalPlayer* LocalPlayer = GetLocalPlayer();
73 if (LocalPlayer == nullptr)
74 {
75 if (UGameInstance* GameInstance = World->GetGameInstance())
76 {
77 LocalPlayer = GameInstance->GetFirstGamePlayer();
78 }
79
80 if (LocalPlayer == nullptr)
81 {
82 PRINTLOG(TEXT("[LoadingTransitionManager] LocalPlayer is null"));
83 return;
84 }
85 }
86
87 if (!TransitionWidgetClass)
88 {
89 PRINTLOG(TEXT("[LoadingTransitionManager] TransitionWidgetClass is null"));
90 return;
91 }
92
93 APlayerController* PC = LocalPlayer->GetPlayerController(World);
94 if (PC == nullptr)
95 return;
96
97 if (!IsValid(TransitionWidget))
98 {
99 if (auto NewWidget = CreateWidget<ULoadingTransitionWidget>(PC, TransitionWidgetClass))
100 {
101 NewWidget->AddToGameViewport(GameLayer::Loading);
102 TransitionWidget = NewWidget;
103 }
104 }
105 else
106 {
107 TransitionWidget->SetOwningLocalPlayer(LocalPlayer);
108
109 if (!TransitionWidget->IsInViewport())
110 {
111 TransitionWidget->AddToGameViewport(GameLayer::Loading);
112 }
113 }
114}

116void ULoadingTransitionManager::ShowLoadingScreen()
117{
118 if (UWorld* World = GetWorld())
119 {
120 EnsureWidgetForWorld(World);
121
122 if (auto DialogManager = UDialogManager::Get(this))
123 {
124 DialogManager->HideToastImmediately();
125 }
126
127 bIsShowing = true;
128
129 if (TransitionWidget)
130 {
131 if (!TransitionWidget->IsInViewport())
132 {
133 TransitionWidget->AddToGameViewport(GameLayer::Loading);
134 }
135
136 TransitionWidget->SetVisibility(ESlateVisibility::Visible);
137
138 LatestReportedProgress = 0.0f;
139 TransitionWidget->RefreshLoadingTip();
140
141 if (World)
142 {
143 World->GetTimerManager().ClearTimer(HideTimerHandle);
144 }
145
146 LastShowTimestamp = FPlatformTime::Seconds();
147 bHideRequested = false;
148 }
149 }
150}

152void ULoadingTransitionManager::HideLoadingScreen()
153{
154 if (!TransitionWidget)
155 return;
156
157 LatestReportedProgress = FMath::Clamp(LatestReportedProgress, 0.0f, 1.0f);
158
159
160 if (UWorld* World = GetWorld())
161 {
162 if (HideTimerHandle.IsValid())
163 {
164 World->GetTimerManager().ClearTimer(HideTimerHandle);
165 }
166
167 const double Elapsed = FPlatformTime::Seconds() - LastShowTimestamp;
168 double Delay = 0.0;
169
170 if (Elapsed < MinVisibleDurationSeconds)
171 {
172 Delay = MinVisibleDurationSeconds - Elapsed;
173 }
174
175 Delay = FMath::Max(Delay, HoldAfterCompletionSeconds);
176
177 if (Delay > KINDA_SMALL_NUMBER)
178 {
179 bHideRequested = true;
180 World->GetTimerManager().SetTimer(
181 HideTimerHandle,
182 this,
183 &ULoadingTransitionManager::FinalizeHide,
184 Delay,
185 false);
186 return;
187 }
188 }
189
190 FinalizeHide();
191
192}

194void ULoadingTransitionManager::FinalizeHide()
195{

197 {
198 if (HideTimerHandle.IsValid())

200 World->GetTimerManager().ClearTimer(HideTimerHandle);
201 }

203
204 if (TransitionWidget)
205 {
206 if (TransitionWidget->IsInViewport())
207 {
208 TransitionWidget->RemoveFromParent();
209 }
210
211 TransitionWidget->SetVisibility(ESlateVisibility::Collapsed);
212 }
213
214 bHideRequested = false;
215 bIsShowing = false;
216}

257bool ULoadingTransitionManager::DoesWorldBelongToLocalPlayer(const UWorld* World) const
258{
259 if (!World)
260 return false;
261
262 if (const ULocalPlayer* LocalPlayer = GetLocalPlayer())
263 {
264 return LocalPlayer->GetGameInstance() == World->GetGameInstance();
265 }
266
267 return false;
