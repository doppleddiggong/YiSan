107void AQuestManagerActor::NextQuest()
108{
109 if (!HasAuthority())
110 return;
111
112 if (!QuestList.IsValidIndex(QuestIndex))
113 {
114 PRINTLOG(TEXT("[QUEST] NextQuest called with invalid index"));
115 QuestTarget = EBuildingType::None;
116 SendUpdateQuest();
117 return;
118 }
119
120 ++QuestIndex;
121 QuestTarget = QuestList.IsValidIndex(QuestIndex) ? QuestList[QuestIndex] : QuestTarget = EBuildingType::None;
122 SendUpdateQuest();
123}

126void AQuestManagerActor::ContactBuilding(EBuildingType InType)
127{
128 if (!HasAuthority())
129 return;
130
131 if (!HasActiveQuest())
132 {
133 PRINTLOG(TEXT("[QUEST] Contact ignored. No active quest."));
134 return;
135 }
136
137 if (InType != QuestTarget)
138 {
139 PRINTLOG(TEXT("[QUEST] Contact %s ignored. Current target is %s"),
140 *ENUM_TO_NAME(EBuildingType, InType), *ENUM_TO_NAME(EBuildingType, QuestTarget));
141 return;
142 }
143
144 PRINTLOG(TEXT("[QUEST] Contact confirmed for %s"), *ENUM_TO_NAME(EBuildingType, QuestTarget));
145 NextQuest();
146}
