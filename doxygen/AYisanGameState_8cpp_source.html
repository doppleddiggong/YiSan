151void AYisanGameState::UpdatePlayerList()
152{
153 if (!HasAuthority())
154 return;
155
156 // Sort PlayerArray by PlayerIndex
157 PlayerArray.Sort([](const APlayerState& A, const APlayerState& B)
158 {
159 const AYiSanPlayerState* PSA = Cast<const AYiSanPlayerState>(&A);
160 const AYiSanPlayerState* PSB = Cast<const AYiSanPlayerState>(&B);
161
162 if (!PSA || !PSB)
163 {
164 return A.GetPlayerName() < B.GetPlayerName();
165 }
166
167 return PSA->PlayerIndex < PSB->PlayerIndex;
168 });
169
170 TArray<FString> PlayerInfoList;
171
172 for (int32 i = 0; i < PlayerArray.Num(); ++i)
173 {
174 if (auto PS = Cast<AYiSanPlayerState>(PlayerArray[i]))
175 {
176 PlayerInfoList.Add(FString::Printf(TEXT("%d:%s"), PS->PlayerIndex, *PS->Nickname));
177 }
178 }
179
180 PlayerList = PlayerInfoList;
181 PRINTLOG(TEXT("AYisanGameState: PlayerList updated. Current players: %s"),
182 *FString::Join(PlayerList, TEXT(", ")));
183
184 OnRep_PlayerList();
185}

188void AYisanGameState::OnRep_PlayerList()
189{
190 // Notify NetworkSubsystem on clients
191 if (!HasAuthority())
192 {
193 if (UGameInstance* GI = GetGameInstance())
194 {
195 if (UYisanOnlineSystem* NetworkSubsystem = GI->GetSubsystem<UYisanOnlineSystem>())
196 {
197 NetworkSubsystem->OnPlayerListUpdated.Broadcast(PlayerList);
198 }
199 }
200 }
201
202 // Broadcast to direct delegates
203 OnPlayerListUpdated.Broadcast(PlayerList);
204}
