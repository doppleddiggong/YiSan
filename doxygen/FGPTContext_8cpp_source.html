117TSharedPtr<FJsonObject> FGPTContext::ToJsonObject() const
118{
119 if (!HasAnyData())
120 {
121 return nullptr;
122 }
123
124 TSharedPtr<FJsonObject> JsonObject = MakeShared<FJsonObject>();
125
126 if (current_location.IsValid())
127 {
128 if (TSharedPtr<FJsonObject> LocationJson = current_location.ToJsonObject())
129 {
130 JsonObject->SetObjectField(TEXT("current_location"), LocationJson);
131 }
132 }
133
134 if (focused_object.IsValid())
135 {
136 if (TSharedPtr<FJsonObject> FocusJson = focused_object.ToJsonObject())
137 {
138 JsonObject->SetObjectField(TEXT("focused_object"), FocusJson);
139 }
140 }
141
142 TArray<TSharedPtr<FJsonValue>> NearbyArray;
143 for (const FGPTContextNearBuilding& Building : nearby_buildings)
144 {
145 if (!Building.IsValid())
146 {
147 continue;
148 }
149
150 if (TSharedPtr<FJsonObject> BuildingJson = Building.ToJsonObject())
151 {
152 NearbyArray.Add(MakeShared<FJsonValueObject>(BuildingJson));
153 }
154 }
155
156 if (NearbyArray.Num() > 0)
157 {
158 JsonObject->SetArrayField(TEXT("nearby_buildings"), NearbyArray);
159 JsonObject->SetArrayField(TEXT("nearest_buildings"), NearbyArray);
160 }
161
162 return JsonObject;
