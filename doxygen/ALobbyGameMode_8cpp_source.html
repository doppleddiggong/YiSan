39void ALobbyGameMode::PostLogin(APlayerController* NewPlayer)
40{
41 Super::PostLogin(NewPlayer);
42
43 // Lobby is local-only, PlayerIndex will be assigned when joining host server
44 PRINTLOG(TEXT("[LobbyGameMode] PostLogin - %s (PlayerIndex assignment skipped in lobby)"),
45 *GetNameSafe(NewPlayer));
46
47 // Pawn이 없으면 강제로 생성
48 if (NewPlayer && !NewPlayer->GetPawn())
49 {
50 AActor* PlayerStart = FindPlayerStart(NewPlayer);
51 APawn* NewPawn = SpawnDefaultPawnFor(NewPlayer, PlayerStart);
52 if (NewPawn)
53 NewPlayer->Possess(NewPawn);
54 }
55}
