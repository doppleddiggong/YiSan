49void UGPTContextSystem::TickComponent(float DeltaTime, ELevelTick TickType,
50 FActorComponentTickFunction* ThisTickFunction)
51{
52 Super::TickComponent(DeltaTime, TickType, ThisTickFunction);
53
54 if ( Owner == nullptr )
55 return;
56
57 // 로컬 플레이어에 대해서만 GPT 컨텍스트를 업데이트합니다.
58 // Host 클라이언트에서 Guest 플레이어의 GPTContextSystem이 동작하지 않도록 방지합니다.
59 if (!Owner->IsLocallyControlled())
60 return;
61
62 // DeltaTime을 계속 더해줍니다.
63 TimeSinceLastCheck += DeltaTime;
64
65 // TimeSinceLastCheck가 1초를 넘어섰는지 확인합니다.
66 if (TimeSinceLastCheck >= 1.0f)
67 {
68 // 타이머를 리셋합니다. (정확한 주기를 위해 1.0f를 빼줍니다)
69 TimeSinceLastCheck -= 1.0f;
70
71 // 1초 마다 실행될 로직
72 this->CheckBuildingInView();
73 this->FindNearestBuilding();
74 }
75}

78void UGPTContextSystem::CheckBuildingInView()
79{
80 const FVector Start = Owner->FollowCamera->GetComponentLocation();
81 const FVector End = Start + Owner->FollowCamera->GetForwardVector() * 5000.f;
82
83 FHitResult HitResult;
84 FCollisionQueryParams CollisionParams;
85 CollisionParams.AddIgnoredActor(Owner);
86
87 EBuildingType TempBuildingType = EBuildingType::None;
88 ABuilding* HitBuilding = nullptr;
89
90 if (GetWorld()->LineTraceSingleByChannel(HitResult, Start, End, ECC_Visibility, CollisionParams))
91 {
92 if (auto HitActor = HitResult.GetActor())
93 {
94 if (HitActor->ActorHasTag(GameTags::Building))
95 {
96 if (ABuilding* Building = Cast<ABuilding>(HitActor))
97 {
98 TempBuildingType = Building->BuildingType;
99 HitBuilding = Building;
100 }
101 }
102 }
103 }
104
105 const bool bTypeChanged = !FocusBuildingType.IsSet() || FocusBuildingType.GetValue() != TempBuildingType;
106
107 if (TempBuildingType != EBuildingType::None && HitBuilding)
108 {
109 if (bTypeChanged)
110 {
111 FocusBuildingType = TempBuildingType;
112 BroadcastManager->SendFocusBuilding(FocusBuildingType.GetValue());
113 }
114 }
115 else
116 {
117 if (FocusBuildingType.IsSet())
118 {
119 FocusBuildingType.Reset();
120 BroadcastManager->SendFocusBuilding( EBuildingType::None );
121 }
122 }
123}

127void UGPTContextSystem::FindNearestBuilding()
128{
129 auto FoundBuildings = FComponentHelper::GetAllOfClass<ABuilding>(GetWorld());
130
131 float MinDistance = TNumericLimits<float>::Max();
132
133 for (auto Building : FoundBuildings)
134 {
135 float Distance = Owner->GetDistanceTo(Building);
136 if (Distance < MinDistance)
137 {
138 MinDistance = Distance;
139 NearestBuilding = Building;
140 }
141 }
142
143 if (NearestBuilding.Get())
144 {
145 // 근접 했다는 정보로 small popup 을 띄울 예정
146 BroadcastManager->SendNearBuilding(NearestBuilding->BuildingType);
147 }
148 else
149 {
150 // 건물이 없다면 popup 을 닫을예정
151 BroadcastManager->SendNearBuilding(EBuildingType::None);
152 }
153}

168FGPTContext UGPTContextSystem::GetGPTContext() const
169{
170 FGPTContext Context;
171
172 UWorld* World = GetWorld();
173 if (!World)
174 return Context;
175
176 const FVector PlayerLocation = Owner->GetActorLocation();
177 UGameDataManager* DataManager = UGameDataManager::Get(World);
178
179 TArray<FBuildingSnapshot> BuildingSnapshots;
180 BuildingSnapshots.Reserve(32);
181
182 constexpr float CmToMeters = 0.01f;
183
184 for (TActorIterator<ABuilding> It(World); It; ++It)
185 {
186 ABuilding* Building = *It;
187 if (!IsValid(Building))
188 {
189 continue;
190 }
191
192 FBuildingSnapshot Snapshot;
193 Snapshot.Actor = Building;
194 Snapshot.Location = Building->GetActorLocation();
195 Snapshot.DistanceMeters = FVector::Dist(PlayerLocation, Snapshot.Location) * CmToMeters;
196 Snapshot.Name = UGameDataManager::Get(GetWorld())->GetBuildingDataName(Building->BuildingType);
197 if (Snapshot.Name.IsEmpty())
198 {
199 Snapshot.Name = Building->GetName();
200 }
201 BuildingSnapshots.Add(MoveTemp(Snapshot));
202 }
203
204 BuildingSnapshots.Sort([](const FBuildingSnapshot& A, const FBuildingSnapshot& B)
205 {
206 return A.DistanceMeters < B.DistanceMeters;
207 });
208
209 const FString CurLocationName = BuildingSnapshots.Num() > 0 ? BuildingSnapshots[0].Name : FString(TEXT("전하 위치"));
210 Context.current_location.Set(CurLocationName, PlayerLocation);
211
212 if (FocusBuildingType.IsSet())
213 {
214 FBuildingData Params;
215 if (UGameDataManager::Get(GetWorld())->GetBuildingData(FocusBuildingType.GetValue(), Params))
216 {
217 Context.focused_object.Set(Params.name, FVector(Params.x, Params.y, Params.z));
218 }
219 }
220
221 constexpr int32 MaxNearbyBuildings = 4;
222 int32 AddedCount = 0;
223 TSet<FString> UniqueNames;
224
225 for (const FBuildingSnapshot& Snapshot : BuildingSnapshots)
226 {
227 if (Snapshot.Name.IsEmpty())
228 continue;
229
230 if (UniqueNames.Contains(Snapshot.Name))
231 continue;
232
233 FGPTContextNearBuilding Nearby;
234 const float RoundedDistance = FMath::RoundToFloat(Snapshot.DistanceMeters * 10.0f) / 10.0f;
235 Nearby.Set(Snapshot.Name, RoundedDistance);
236 Context.AddNearbyBuilding(Nearby);
237 UniqueNames.Add(Snapshot.Name);
238
239 if (++AddedCount >= MaxNearbyBuildings)
240 break;
241 }
242
243 return Context;
244}
