86void ULobbyWidget::OnHostButtonClicked()
87{
88 UTextBlock* buttonText = Cast<UTextBlock>(Btn_Host->GetChildAt(0));
89
90 Btn_Host->SetVisibility(ESlateVisibility::HitTestInvisible);
91 buttonText->SetText(FText::FromString(TEXT("만드는 중")));
92
93 FText nameText = editSessionName->GetText();
94 FText sizeText = editSessionSize->GetText();
95
96 if (nameText.IsEmpty() || sizeText.IsEmpty()) return;
97
98 FString sessionName = nameText.ToString();
99 int32 sessionSize = FCString::Atoi(*(sizeText.ToString()));
100
101 PRINTLOG(TEXT("[LobbyWidget] OnHostButtonClicked - Map=%s, SessionName=%s, MaxPlayers=%d"), *MapName, *sessionName, MaxPlayers);
102
103 UYisanOnlineSystem::Get(GetWorld())->CreateMySession(sessionName, sessionSize);
104}

117void ULobbyWidget::OnFindComplete(int32 idx, FString sessionName)
118{
119 PRINTLOG(TEXT("[LobbyWidget] OnFindComplete - idx: %d, sessionName: %s"), idx, *sessionName);
120
121 if (sessionName.IsEmpty() == false && idx >= 0)
122 {
123 // sessionInfoWidget 만들자.
124 USessionInfoWidget* item = CreateWidget<USessionInfoWidget>(GetWorld(), sessionInfoWidget);
125 // 만들어진 item 을 scrollSessionList 에 추가
126 scrollSessionList->AddChild(item);
127 // 만들어지 item 정보 설정
128 {
129 item->SetSessionInfo(idx, sessionName);
130 }
131 }
132
133 // idx == -1: 모든 세션 검색 완료 신호
134 if (idx == -1)
135 {
136 SetFindingText(TEXT("방 목록"));
137 Btn_Find->SetVisibility(ESlateVisibility::Visible);
138 ULoadingCircleManager::Get(GetWorld())->Hide();
139 }
140}

171void ULobbyWidget::OnClickName()
172{
173 if (!editNickname) return;
174
175 const FString EnteredName = editNickname->GetText().ToString();
176
177 if (EnteredName.IsEmpty())
178 {
179 UE_LOG(LogTemp, Warning, TEXT("닉네임을 입력해주세요!"));
180 return;
181 }
182
183 UYisanOnlineSystem::Get(GetWorld())->SetPlayerNickname(EnteredName);
184 UE_LOG(LogTemp, Log, TEXT("닉네임 저장됨: %s"), *EnteredName);
185
186 widgetSwitcher->SetActiveWidgetIndex(1);
187}
